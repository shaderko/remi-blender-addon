cmake_minimum_required(VERSION 3.16)
project(remi_adaptive LANGUAGES CXX)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/adaptive.cpp" REMI_SOURCE_SHA256)
add_library(remi_adaptive SHARED adaptive.cpp)
target_compile_features(remi_adaptive PRIVATE cxx_std_17)
target_compile_definitions(remi_adaptive PRIVATE REMI_SOURCE_SHA256="${REMI_SOURCE_SHA256}")
set_target_properties(remi_adaptive PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(remi_adaptive PRIVATE /O2 /EHsc)
  target_compile_definitions(remi_adaptive PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(remi_adaptive PRIVATE -O3)
endif()
install(TARGETS remi_adaptive LIBRARY DESTINATION . RUNTIME DESTINATION .)
